아키텍처적 해결책

1. JSON 구조화 매핑 및 스레드 독립 세션

객실별 Credential을 구조화된 JSON 형태로 관리하여 `get_credentials` 함수를 통해 동적으로 로딩합니다. 병렬 실행 시 Playwright의 `new_context`를 통해 각 세션을 물리적 브라우저 환경처럼 격리하여 데이터 및 세션 충돌을 근본적으로 차단했습니다.

2. 유연한 Locator 스택 및 Stealth 우회

계정 프로필 선택 시 다양한 CSS 셀렉터 폴백(Fallback) 방식을 채택하고, playwright_stealth 및 랜덤 타이핑 딜레이를 통해 봇 탐지 위협 요소를 최소화하여 안정적인 화면 진입을 유지합니다.

class NetflixManager:
    @classmethod
    async def run_language_change(cls, room_number: str, korean_lang: str) -> str:
        user_data = load_data(USER_DATA_PATH, "User Data")
        lang_map = load_data(LANGUAGE_MAP_PATH, "Language Map")

        # 객실에 대응하는 넷플릭스 크레덴셜 조회 (다중 아이디 라우팅)
        email, password = get_credentials(user_data, room_number)
        if not (email and password):
            return f"Room number '{room_number}' not found."

        target_lang = get_language_code(korean_lang, lang_map)
        
        async with async_playwright() as p:
            browser = await p.chromium.launch(headless=True)
            # 브라우저 컨텍스트 격리 실행
            context = await browser.new_context(user_agent="Mozilla/5.0 ...")
            page = await context.new_page()

            # 프로필 관리 페이지 이동 후 마지막 프로필 선택 및 언어 변경 적용
            if await cls._login(page, email, password):
                if await cls._select_last_profile(page):
                    if await cls._change_language_direct(page, target_lang):
                        await browser.close()
                        return f"Success: Room {room_number} to {korean_lang}"
            await browser.close()
            return "Failed to change language"

# 사용자 입력을 Queue를 통해 백엔드 브라우저 스레드로 전달하는 로직

active_verifications = {} # 대화방 ID별 동적 Queue 관리 맵

async def select_room_verify(update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
    room = query.data.replace("v_room_", "")
    chat_id = update.effective_chat.id
    
    info_q = queue.Queue()
    code_q = queue.Queue()
    active_verifications[chat_id] = {"info_q": info_q, "code_q": code_q}

    # 백엔드 스레드로 Playwright 조작 루틴 분기 실행
    context.application.create_task(
        run_verification_task(context, chat_id, room, info_q, code_q)
    )
    return ENTER_NAME

async def enter_sms_code(update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
    sms_code = update.message.text.strip()
    chat_id = update.effective_chat.id
    
    if chat_id in active_verifications:
        # Playwright 스레드가 대기 중인 Queue로 값을 전달하여 블로킹 상태 해제
        active_verifications[chat_id]["code_q"].put(sms_code)
